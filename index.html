Production Backend & Database Blueprint

PostgreSQL / Supabase Schema + Node.js / Express Dynamic NFC Redirect

1. schema.sql (PostgreSQL / Supabase) 
CREATE TABLE businesses (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  slug TEXT UNIQUE NOT NULL,
  name TEXT NOT NULL,
  city TEXT DEFAULT 'Sfax, Tunisia',
  tagline TEXT NOT NULL,
  theme_color TEXT DEFAULT '#111114',
  google_place_id TEXT NOT NULL,
  instagram_handle TEXT,
  whatsapp_number TEXT,
  is_active BOOLEAN DEFAULT TRUE,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE cards (
  card_code VARCHAR(10) PRIMARY KEY, -- e.g. 'A001' written to NTAG215
  business_id UUID REFERENCES businesses(id) ON DELETE SET NULL,
  label TEXT NOT NULL,               -- e.g. 'Cashier Desk', 'Waiter 1'
  created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE analytics_events (
  id BIGSERIAL PRIMARY KEY,
  business_id UUID REFERENCES businesses(id) ON DELETE CASCADE,
  card_code VARCHAR(10) REFERENCES cards(card_code),
  event_type TEXT NOT NULL,          -- 'tap', 'google_click', 'instagram_click', 'whatsapp_click'
  created_at TIMESTAMPTZ DEFAULT NOW()
);
2. server.js (Express Dynamic NFC Router) 
// Every physical NFC card is locked with: https://tappulse.tn/t/:cardCode
app.get('/t/:cardCode', async (req, res) => {
  const { cardCode } = req.params;
  const card = await db.query(
    `SELECT c.card_code, b.* FROM cards c
     JOIN businesses b ON c.business_id = b.id
     WHERE c.card_code = $1 AND b.is_active = TRUE`,
    [cardCode.toUpperCase()]
  );

  if (!card.rows.length) return res.status(404).send('Card not activated');
  const venue = card.rows[0];

  // 1. Log the physical NFC tap asynchronously
  await db.query(
    `INSERT INTO analytics_events (business_id, card_code, event_type) VALUES ($1, $2, 'tap')`,
    [venue.id, cardCode.toUpperCase()]
  );

  // 2. Render or redirect to the venue's branded landing page
  res.redirect(`/c/${venue.slug}?card=${cardCode.toUpperCase()}`);
});
